package object

var sinkOrder []string
var sinkRecords []Record

func resetSinks() { sinkOrder = nil; sinkRecords = nil }
func SendWebhooks(record *Record) error {
	sinkOrder = append(sinkOrder, "webhook")
	sinkRecords = append(sinkRecords, *record)
	return nil
}
func addRecord(record *Record) (int64, error) {
	sinkOrder = append(sinkOrder, "database")
	sinkRecords = append(sinkRecords, *record)
	return 1, nil
}
func sendAuditRecord(record *Record) {
	sinkOrder = append(sinkOrder, "audit")
	sinkRecords = append(sinkRecords, *record)
}
